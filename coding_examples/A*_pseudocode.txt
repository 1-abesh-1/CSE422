A_STAR(start):
    OPEN   ← priority queue ordered by f(n), lowest first
    CLOSED ← empty set

    insert start into OPEN        // g(start) = 0, f(start) = h(start)

    while OPEN is not empty:
        n ← remove node with lowest f(n) from OPEN

        if goalTest(n) then
            return path from start to n    // follow parent pointers

        add n to CLOSED

        for each successor m of n:
            if m is in CLOSED:
                continue

            // g(m) is the path cost to m via n
            // f(m) = g(m) + h(m)

            if m is not in OPEN:
                set parent(m) ← n
                insert m into OPEN with priority f(m)

            else if the new g(m) (via n) is less than the g(m) stored in OPEN:
                set parent(m) ← n
                update m in OPEN with the new g(m) and f(m)

    return FAILURE    // OPEN is empty, no path exists
